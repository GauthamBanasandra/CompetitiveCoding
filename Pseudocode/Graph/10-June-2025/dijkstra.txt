using Node = int
using Weight = int
using AdjList = unordered_map<Node, vector<pair<Node, Weight>>>

Node source
AdjList adj_list;
unordered_map<Node, Weight> min_dist
priority_queue<pair<Weight, Node>, vector<pair<Weight, Node>>, greater<>> order
vector<Node> vertices

for (u : vertices) {
    min_dist[u] = inf
}

min_dist[u] = 0
order.emplace(0, source)
while (!order.empty()) {
    [du, u] = order.top()
    order.pop()

    if (du > min_dist[u]) {
        continue
    }

    for (v : adj_list[u]) {
        if (dv + min_dist[u] < min_dist[v]) {
            min_dist[v] = dv + min_dist[u]
            order.emplace(min_dist[v], v)
        }
    }
}

return min_dist[v]
