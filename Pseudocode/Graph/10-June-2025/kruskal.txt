using Node = int
using Weight = int
using EdgeList = vector<pair<Weight, pair<Node, Node>>>

num_vertices
UFDS ufds(num_vertices)
EdgeList edge_list;
ranges::sort(edge_list)

cost = 0
for ([w, uv], edge_list) {
    [u, v] = uv
    if (!ufds.IsSameSet(u, v)) {
        cost += w
        ufds.union(u, v)
    }
}

return cost
