using Node = int
using Weight = int
using Edge = pair<Node, Node>

vector<pair<Weight, Edge>> edge_list
num_vertices
UFDS ufds(num_vertices)

ranges::sort(edge_list)

cost = 0
for ([w, edge] : edge_list) {
    [u, v] = edge
    if (!ufds.IsSameSet(u, v)) {
        cost += w
        ufds.Union(u, v)
    }
}
